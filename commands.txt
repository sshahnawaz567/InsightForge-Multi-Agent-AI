# Viewing resources
kubectl get pods -n insightforge
kubectl get services -n insightforge
kubectl get deployments -n insightforge
kubectl get all -n insightforge

# Describing resources (detailed info)
kubectl describe pod <pod-name> -n insightforge
kubectl describe service backend-service -n insightforge

# Logs
kubectl logs -n insightforge <pod-name>
kubectl logs -n insightforge -l app=backend  # All backend pods
kubectl logs -n insightforge <pod-name> -f   # Follow

# Exec into pod (like docker exec)
kubectl exec -it -n insightforge <pod-name> -- bash
kubectl exec -it -n insightforge <pod-name> -- python

# Port forwarding
kubectl port-forward -n insightforge service/backend-service 8080:80

# Scaling manually
kubectl scale deployment backend -n insightforge --replicas=5

# Delete resources
kubectl delete pod <pod-name> -n insightforge
kubectl delete -f k8s/  # Delete all
kubectl delete namespace insightforge  # Delete everything!

# Update deployment (after code change)
kubectl rollout restart deployment backend -n insightforge
kubectl rollout status deployment backend -n insightforge
kubectl rollout undo deployment backend -n insightforge  # Rollback

minikube stop      # Stop cluster
minikube delete    # Delete cluster
minikube dashboard # Open web UI
minikube ssh       # SSH into node
minikube logs      # View logs


# Start services
docker-compose up -d

# Stop services
docker-compose down

# Stop and remove volumes (DELETES DATA!)
docker-compose down -v

# Restart a service
docker-compose restart backend

# View logs
docker-compose logs -f backend

# Execute command in container
docker-compose exec backend bash
docker-compose exec postgres psql -U postgres -d insightforge

# See resource usage
docker stats

# Clean up unused images
docker system prune -a

# See disk usage
docker system df
